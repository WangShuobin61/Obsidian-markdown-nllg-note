导航命令
命令1：整车启动 roslaunch c8_navigation real_robot_rviz.launch
命令2：不启动机械臂 roslaunch robot_driver diff_driver.launch
命令3：启动键盘和手柄 roslaunch robot_driver diff_driver1.launch
命令4：启动RViz rosrun rviz rviz -d $(rospack find c8_navigation)/rviz/c8_navigation.rviz
命令5：启动建图 roslaunch c8_navigation gma_only.launch
命令6：保存地图 roslaunch c8_navigation save_map.launch map_path:=/home/wu/maps/my_map
命令7：已有地图定位导航 roslaunch c8_navigation navigation.launch map_file:=/home/wu/maps/my_map.yaml &&(此处需要延迟3秒) rosrun rviz rviz -d $(rospack find c8 navigation)/rviz/c8 navigation.rviz
